% CONTENT_PLACEHOLDER

%---------------------------------------------------------------
% COLORS
% heading  #111111  near-black  → section titles, bold labels, name
% body     #333333  dark gray   → main paragraph text
% lighttext#666666  mid gray    → dates, locations, subtitles
% linkcolor#003366  dark blue   → all hyperlinks
%---------------------------------------------------------------
\definecolor{heading}{HTML}{111111}
\definecolor{body}{HTML}{333333}
\definecolor{lighttext}{HTML}{666666}
\definecolor{linkcolor}{HTML}{003366}

\hypersetup{
  colorlinks=true,
  urlcolor=linkcolor,
  linkcolor=linkcolor
}

%---------------------------------------------------------------
% GLOBAL TEXT SETTINGS
% • No paragraph indent
% • 1pt paragraph skip (tight but not cramped)
% • 1.05 line stretch (slight breathing room)
% • List items: 1pt spacing, 1.8em left margin
%---------------------------------------------------------------
\setlength{\parindent}{0pt}
\setlength{\parskip}{1pt}
\renewcommand{\baselinestretch}{1.05}

\setlist[itemize]{
  leftmargin=1.8em,
  itemsep=1pt,
  topsep=1pt
}

%---------------------------------------------------------------
% SECTION FORMATTING
% • \large\bfseries in heading color
% • 0pt left indent, 6pt before, 2pt after
% • Always use \section* (no numbering)
% • Always follow with \sectrule
%---------------------------------------------------------------
\usepackage{titlesec}

\titleformat{\section}{
  \large\bfseries\color{heading}
}{}{0pt}{}

\titlespacing*{\section}{0pt}{6pt}{2pt}

%---------------------------------------------------------------
% CUSTOM COMMANDS
%
% \sectrule  → thin horizontal rule used after every section header
%              usage: \section*{TITLE} ... content ... \sectrule
%
% \name{X}   → Huge bold heading-color name at top of document
%---------------------------------------------------------------
\newcommand{\sectrule}{\vspace{2pt}\hrule\vspace{3pt}}

\newcommand{\name}[1]{%
  {\Huge\bfseries\color{heading} #1}\par\vspace{4pt}
}

%---------------------------------------------------------------
% SECTION CONTENT PATTERNS
%
% ── Plain paragraph section (Profile, Key Competencies) ──────
% \section*{SECTION TITLE}
% {\color{body}
%   Text here.\par
%   {\color{heading}\textbf{Sub-label:}} More text.
% }
% \vspace{2pt}
% \sectrule
%
% ── Entry with date (Education, Work Experience, Projects) ───
% {\color{heading}\textbf{Title}} \hfill
% {\color{lighttext}Date range}\par
% {\color{body}\textit{Subtitle / tech stack / institution}}\par
% {\color{body}
%   Description text.
% }
% \vspace{3pt}   ← between entries; \vspace{2pt}\sectrule after last
%
% ── Skills line (bold label + body text on same line) ────────
% {\color{heading}\textbf{Category:}} {\color{body}Skill A; Skill B; Skill C.}\par
%
% ── Key Competencies (inline bullets) ────────────────────────
% {\color{body}
%   {\color{heading}\textbf{Category:}} item, item, item.\par
%   {\color{heading}\textbf{Category:}} item, item, item.
% }
%
% ── Bullet-list entries (Projects with detail) ───────────────
% {\color{heading}\textbf{Project Title}} \hfill {\color{lighttext}Date}\par
% {\color{body}\textit{Tech stack}}\par
% \begin{itemize}
%   \item First point.
%   \item Second point.
% \end{itemize}
%---------------------------------------------------------------

%---------------------------------------------------------------
% SPACING RULES SUMMARY
% \vspace{3pt}  between entries within a section
% \vspace{2pt}  before \sectrule at end of section
% \vspace{2pt}\sectrule  closes every section
% \vspace{4pt}  after \name{}
% \vspace{3pt}  after header block before first \sectrule
%---------------------------------------------------------------
